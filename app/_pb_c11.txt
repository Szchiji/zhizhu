        tenant.bot_token_enc = encrypt_token(text)
        db.commit()
        wh = ""
        try:
            from app.config import WEBHOOK_BASE_URL, WEBHOOK_SECRET
            if WEBHOOK_BASE_URL:
                wh = f"{WEBHOOK_BASE_URL.rstrip('/')}/wh/t/{tenant.id}"
                async with httpx.AsyncClient(timeout=20) as client:
                    wr = await client.post(
                        f"https://api.telegram.org/bot{text}/setWebhook",
                        json={
                            "url": wh,
                            "secret_token": WEBHOOK_SECRET,
                            "allowed_updates": ["message", "callback_query", "inline_query"],
                        },
                    )
                if not (wr.json() or {}).get("ok"):
                    await update.effective_message.reply_text(
                        f"Token 已保存 @{tenant.bot_username}，但Webhook设置失败：{(wr.json() or {}).get('description')}"
                    )
                    return
        except Exception as exc:
            await update.effective_message.reply_text(f"Token 已保存，Webhook异常：{exc}")
            return
        tip = f"\nWebhook: {wh}" if wh else "\n(未配置 WEBHOOK_BASE_URL，请手动设Webhook到 /wh/t/{id})"
        await update.effective_message.reply_text(f"已克隆 @{tenant.bot_username}{tip}")
    finally:
        db.close()


async def cmd_admin_alias(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    await cmd_admin(update, context)


def build_platform_app(token: str) -> Application:
    app = Application.builder().token(token).updater(None).build()
    for name, fn in [
        ('start', cmd_start), ('status', cmd_status), ('help', cmd_help), ('admin', cmd_admin),
        ('paid', cmd_paid), ('bind', cmd_bind),
        ('confirm', cmd_admin_alias), ('prices', cmd_admin_alias), ('setprice', cmd_admin_alias),
        ('setaddr', cmd_admin_alias), ('clone', cmd_admin_alias),
    ]:
        app.add_handler(CommandHandler(name, fn))
    app.add_handler(InlineQueryHandler(on_inline))
    app.add_handler(CallbackQueryHandler(on_callback))
    app.add_handler(PreCheckoutQueryHandler(on_precheckout))
    app.add_handler(MessageHandler(filters.StatusUpdate.WEB_APP_DATA, on_webapp))
    app.add_handler(MessageHandler(filters.SUCCESSFUL_PAYMENT, on_successful_payment))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, on_text))
    return app
